                cids=cids,
                peer_id="12D3KooWEicpP6sEUkd3tJYKMAU4qHFJ2RWvxg5NViC5QCVK6wef",
                api_url="http://127.0.0.1:5001"
            )

                logger.info("CID: %s, Block CID: %s, Peer has block: %s, Error: %s",

            logger.error("Error in main: %s", e)

    asyncio.run(main())
